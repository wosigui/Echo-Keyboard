cmake_minimum_required(VERSION 3.22.1)
project(echo_pinyin)
file(GLOB PINYIN_SOURCES pinyin/share/*.cpp)
add_library(echo_pinyin SHARED echo_jni.cpp ${PINYIN_SOURCES})
target_include_directories(echo_pinyin PRIVATE pinyin/include)
target_compile_features(echo_pinyin PRIVATE cxx_std_17)
target_compile_options(echo_pinyin PRIVATE -Wno-register -Wno-deprecated -Wno-format)
target_link_options(echo_pinyin PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(echo_pinyin log)
